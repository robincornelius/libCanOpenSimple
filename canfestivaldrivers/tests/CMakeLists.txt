cmake_minimum_required(VERSION 3.20)
project(CanUsbParserTests LANGUAGES CXX)
enable_testing()
add_executable(canusb_parser_tests CanUsbParserTests.cpp)
target_compile_features(canusb_parser_tests PRIVATE cxx_std_11)
if(MSVC)
    target_compile_options(canusb_parser_tests PRIVATE /W4 /WX)
else()
    target_compile_options(canusb_parser_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME canusb_parser_tests COMMAND canusb_parser_tests)

if(WIN32)
    add_library(startup_test_dependency SHARED StartupDependency.cpp)
    add_library(startup_test_driver SHARED StartupDriver.cpp)
    target_link_libraries(startup_test_driver PRIVATE startup_test_dependency)
    add_executable(null_driver_tests NullDriverTests.cpp)
    target_include_directories(null_driver_tests PRIVATE .. ../win32)
    target_compile_features(null_driver_tests PRIVATE cxx_std_11)
    add_test(NAME null_driver_tests COMMAND null_driver_tests)
    set_tests_properties(null_driver_tests PROPERTIES TIMEOUT 10)
    add_executable(serial_overlapped_tests SerialOverlappedTests.cpp)
    target_compile_features(serial_overlapped_tests PRIVATE cxx_std_11)
    if(MSVC)
        target_compile_options(serial_overlapped_tests PRIVATE /W4 /WX)
    else()
        target_compile_options(serial_overlapped_tests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME serial_overlapped_tests COMMAND serial_overlapped_tests)
    set_tests_properties(serial_overlapped_tests PROPERTIES TIMEOUT 20)
endif()
